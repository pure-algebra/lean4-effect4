
/- Isolated finite control: all theorem declarations added by this source file.
   UInt64 Expr.hash agreement is a finite fingerprint comparison, not proof of equality. -/
run_cmd do
  let env ← Lean.getEnv
  let mut count : Nat := 0
  for (name, info) in env.constants.toList do
    if (env.getModuleIdxFor? name).isNone then
      if let .thmInfo thm := info then
        Lean.logInfo m!"A3_PROOF|{name}|{thm.type.hash}|{thm.value.hash}"
        count := count + 1
  Lean.logInfo m!"A3_PROOF_COUNT|{count}"
