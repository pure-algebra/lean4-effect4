-- Proposed proof reduces cleared -> clearStack -> MaskChain [] to the same equality.
theorem cleared_maskChain_iff (base : Bool)
    (interp : RunInterp ν σ β ε δ ι α χ St)
    (f : RunFiber ν σ β ε δ ι α χ) :
    Effect4.FrameFiber.MaskChain base (f.cleared interp).frame.interruptible
      (f.cleared interp).frame.stack ↔ f.frame.interruptible = base := by
  rfl
