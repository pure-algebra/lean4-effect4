663:     valueEffect: Effect.Effect<A, E, any>

670:   await(): Effect.Effect<A, E> {
671:     const exit = this.fiber.pollUnsafe()
672:     if (exit) return exit
673:     this.awaiters++
674:     return effect.onExit(effect.fiberJoin(this.fiber), () => {
675:       this.awaiters--
676:       if (this.awaiters > 0 || this.fiber.pollUnsafe()) return effect.void
677:       // Detach before interrupting so new lookups do not join the abandoned fiber
678:       // while its finalizers run.
679:       this.onInterrupt?.()

686:   const current = MutableHashMap.get(self.map, key)

688:     MutableHashMap.remove(self.map, key)

693:   if (entry.expiresAt === undefined) {
