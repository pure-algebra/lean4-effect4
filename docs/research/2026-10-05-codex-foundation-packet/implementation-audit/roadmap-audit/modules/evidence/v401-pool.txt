679: const releaseItem = <A, E>(self: Pool<A, E>, item: PoolItem<A, E>): Effect.Effect<void> =>
680:   core.withFiber((fiber) => {
681:     const state = self.state
682:     item.refCount--
683:     state.usage--
684:     if (item.refCount === 0) {
685:       item.idleSince = (self as PoolImpl<A, E>)[PoolClock].currentTimeMillisUnsafe()

954: const invalidatePoolItem = <A, E>(self: Pool<A, E>, poolItem: PoolItem<A, E>): Effect.Effect<void> =>
955:   Effect.suspend(() => {
956:     if (!self.state.items.has(poolItem)) {
957:       return Effect.void
958:     } else if (poolItem.refCount === 0) {
959:       self.state.items.delete(poolItem)
960:       removeAvailable(self, poolItem)
961:       self.state.invalidated.delete(poolItem)
962:       return Effect.asVoid(Effect.flatMap(
963:         poolItem.finalizer,
964:         () => Effect.forkIn(Effect.interruptible(resize(self)), self.state.scope, { startImmediately: true })
965:       ))
966:     }
967:     self.state.invalidated.add(poolItem)
968:     removeAvailable(self, poolItem)
969:     // An invalidated item stops counting towards the pool's active size, so the
970:     // pool is now below target and has to top itself back up. Waiting for the
971:     // last lease to be returned would strand anybody already queued: the item
972:     // they are waiting for is never coming back.
973:     return Effect.asVoid(
974:       Effect.forkIn(Effect.interruptible(resize(self)), self.state.scope, { startImmediately: true })
