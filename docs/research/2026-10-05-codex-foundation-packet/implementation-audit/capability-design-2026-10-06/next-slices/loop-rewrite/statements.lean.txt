-- CORE: imports Effect4.Program.Fold only.
namespace Effect4.Program

def stripSuspendsAlg (Op : Type) : EffAlgebra Op (EffSelfCarrier Op) :=
  { EffAlgebra.id Op with eff_suspend := fun body => body }

def stripSuspends {Op : Type} (e : Eff Op) : Eff Op :=
  cata_eff (stripSuspendsAlg Op) e

end Effect4.Program

-- Helpers of looped-suspension-cleanup; public API checks Looped before applying its law.
-- Initial-algebras-folds, compatibility, serving R8 through the named transformation.
@[semantics "initial-algebras-folds" (requirement := R8)]
proof_goal stripSuspends_looped (e : NativeEff) :
  Looped (stripSuspends e) = Looped e

@[semantics "initial-algebras-folds" (requirement := R8)]
proof_goal stripSuspends_straight (e : NativeEff) :
  Straight (stripSuspends e) = Straight e

-- Stronger internal helper, from the existing denotation algebra's homomorphism.
-- Keeping Looped in this public-facing helper prevents fallback equality misuse.
@[semantics "translation-simulation" (requirement := R8)]
proof_goal stripSuspends_denoteB (k : Nat) (e : NativeEff)
    (hl : Looped e = true) (env : List Val) :
  denoteB k (stripSuspends e) env = denoteB k e env

-- Proposed registry pointer: translation-simulation / R8 / role simulation.
-- One semantic budget, all environments and all complete Stores, including unfinished writes.
@[semantics "translation-simulation" (requirement := R8)]
proof_goal looped_suspension_cleanup (e : NativeEff) (hl : Looped e = true) :
  Looped (stripSuspends e) = true ∧
    ∀ (k : Nat) (env : List Val) (stores : Stores),
      meaningB k (stripSuspends e) env stores = meaningB k e env stores

-- Existing StraightEq reused, not a parallel relation with a different observation.
@[semantics "initial-algebras-folds" (requirement := R8)]
@[semantics "translation-simulation" (requirement := R8)]
proof_goal stripSuspends_straightEq (e : NativeEff) (hs : Straight e = true) :
  StraightEq (stripSuspends e) e

-- Derived consumer: reuse loopAgreement at each program, with separate fuel bounds.
-- This exact Api.run statement uses its default empty host table and initial state.
@[semantics "translation-simulation" (requirement := R8)]
proof_goal stripSuspends_completed_run (e : NativeEff) (hl : Looped e = true)
    (k : Nat) (ex : ExitV) (stores : Stores)
    (done : meaningB k e [] Stores.empty = (some ex, stores)) :
  ∃ originalBound cleanedBound : Nat,
    ∀ originalFuel cleanedFuel : Nat,
      originalBound ≤ originalFuel → cleanedBound ≤ cleanedFuel →
      (Api.run e originalFuel).outcome = .finished ∧
      (Api.run (stripSuspends e) cleanedFuel).outcome = .finished ∧
      (Api.run e originalFuel).exit = some ex ∧
      (Api.run (stripSuspends e) cleanedFuel).exit = some ex ∧
      (Api.run e originalFuel).stores = stores ∧
      (Api.run (stripSuspends e) cleanedFuel).stores = stores

end Effect4.Program.Denote
