def performTermWith {Op : Type} (mk : Term → Op)
    (request : TermSrc) (body : TermSrc → TermSrc) : Src Op :=
  fun env path =>
    let current := env.mint "current"
    performTerm mk current (body (minted current)) request env path

-- Intended proof home: Laws.Program.Authoring, beside performTerm_scoped.
-- This follows foldWith_scoped's pointwise proof shape; it asserts scope only.
theorem performTermWith_scoped {Op : Type} [ScopedOp Op] {mk : Term → Op}
    (hmk : ∀ t n, ScopedOp.scopedAt (mk t) n = t.scoped (n + 1))
    {body : TermSrc → TermSrc} {request : TermSrc}
    (hb : ∀ current : TermSrc, current.Scoped → (body current).Scoped)
    (hr : request.Scoped) : (performTermWith mk request body).Scoped :=
  ⟨fun env path term accepted =>
    (performTerm_scoped hmk (env.mint "current")
      (hb _ (minted_scoped _)) hr).holds env path term accepted⟩

def Ref.updateWith (request : TermSrc) (body : TermSrc → TermSrc) : Src NativeOp :=

def noteWith (log x : TermSrc) : Src NativeOp :=
