#!/usr/bin/env python3
"""Seat BOUNDS, assignment 6: the finite model against tsgo.

    python3 model_ts.py gen  <out-folder>     writes model.ts and model-lines.tsv from model-ts.tsv
    python3 model_ts.py join <out-folder>     reads tsgo2-<prelude>-model.txt and prints the tables

`model-ts.tsv` is written by the probe `ModelTs`: one row for each application of six atoms of
the finite model, with the verdict of the present match, of the match by bounds, and of the
match by bounds under each of three guards. `gen` prints each application as one call, at
declared constants of the argument types, with two more lines where the match by bounds
answers: tsgo's type of the call assigned to Lean's answer, and Lean's answer assigned to
tsgo's type. `join` reads tsgo's errors by line, for each of the four preludes of run-ts2.sh.
"""
import collections
import csv
import os
import re
import sys

MODE, OUT = sys.argv[1], sys.argv[2]
PRELUDES = ["present", "split", "cross", "whole"]
csv.field_size_limit(10 ** 9)


def rows():
    with open(os.path.join(OUT, "model-ts.tsv"), newline="") as f:
        return list(csv.DictReader(f, delimiter="\t", quoting=csv.QUOTE_NONE))


def gen():
    data = rows()
    consts = collections.OrderedDict()
    for r in data:
        for t in r["arguments as TypeScript"].split(" ;; "):
            consts.setdefault(t, f"x{len(consts)}")
    lines = [
        "// GENERATED by model_ts.py from model-ts.tsv (seat BOUNDS); one application of the finite model on each `a` line",
        'import { Option, Ref } from "effect"',
        'import { getOrElse, ite, cons, append, get, mapFromEntries } from "./prelude.ts"',
        "export type Used = Option.Option<never> | Ref.Ref<never>",
    ]
    for t, name in consts.items():
        lines.append(f"declare const {name}: {t}")
    index = []
    for k, r in enumerate(data):
        args = ", ".join(consts[t] for t in r["arguments as TypeScript"].split(" ;; "))
        lines.append(f"export const a{k} = {r['atom']}({args})")
        index.append((len(lines), k, "call"))
        if r["bounds"] == "true":
            lines.append(f"export const b{k}: {r['bounds answer']} = a{k}")
            index.append((len(lines), k, "tsgo below Lean"))
            lines.append(f"export const c{k}: typeof a{k} = null! as {r['bounds answer']}")
            index.append((len(lines), k, "Lean below tsgo"))
    with open(os.path.join(OUT, "model.ts"), "w") as f:
        f.write("\n".join(lines) + "\n")
    with open(os.path.join(OUT, "model-lines.tsv"), "w") as f:
        f.write("line\trow\trole\n")
        for line, k, role in index:
            f.write(f"{line}\t{k}\t{role}\n")
    print(f"model.ts: {len(data)} applications, {len(consts)} argument types, {len(lines)} lines")


def table(title, header, body):
    print(f"\n{title}\n")
    print("| " + " | ".join(header) + " |")
    print("| " + " | ".join("---" for _ in header) + " |")
    for row in body:
        print("| " + " | ".join(str(c) for c in row) + " |")


def join():
    data = rows()
    role_of = {}
    with open(os.path.join(OUT, "model-lines.tsv")) as f:
        next(f)
        for line in f:
            n, k, role = line.rstrip("\n").split("\t")
            role_of[int(n)] = (int(k), role)
    # tsgo's verdicts: for each prelude, the rows whose call is refused, and the two type checks
    verdict = {}
    for p in PRELUDES:
        path = os.path.join(OUT, f"tsgo2-{p}-model.txt")
        if not os.path.exists(path):
            print(f"{p}: not run")
            continue
        text = open(path).read()
        bad = collections.defaultdict(set)
        other = 0
        for m in re.finditer(r"^model\.ts\((\d+),\d+\): error (TS\d+)", text, flags=re.M):
            n = int(m.group(1))
            if n in role_of:
                k, role = role_of[n]
                bad[role].add(k)
            else:
                other += 1
        errors = len(re.findall(r"error TS", text))
        verdict[p] = bad
        print(f"- prelude {p}: {errors} error lines; outside the applications: {other}; "
              f"calls refused: {len(bad['call'])} of {len(data)}")
    lean_cols = [("present", "present"), ("bounds", "bounds"),
                 ("bounds, guard of one parameter", "bounds + guard of one parameter"),
                 ("bounds, guard of a split parameter", "bounds + guard of a split parameter"),
                 ("bounds, guard with cross terms", "bounds + guard with cross terms")]
    print(f"\nApplications: {len(data)}; at a normal request: {sum(r['reading'] == 'normal' for r in data)}; "
          f"at a raw request that is not normal: {sum(r['reading'] == 'raw' for r in data)}")
    for col, name in lean_cols:
        print(f"- Lean, {name}: admits {sum(r[col] == 'true' for r in data)}")
    for p in verdict:
        print(f"- tsgo, prelude {p}: accepts {len(data) - len(verdict[p]['call'])}")
    # every Lean reading against every prelude
    body = []
    for col, name in lean_cols:
        for p in verdict:
            both = lean_only = ts_only = neither = 0
            for k, r in enumerate(data):
                lean = r[col] == "true"
                ts = k not in verdict[p]["call"]
                both += lean and ts
                lean_only += lean and not ts
                ts_only += ts and not lean
                neither += (not lean) and (not ts)
            body.append([name, p, both, lean_only, ts_only, neither])
    table("Each reading of the checker against each prelude (all six atoms):",
          ["Lean", "prelude", "both admit", "Lean admits, tsgo refuses", "tsgo accepts, Lean refuses", "both refuse"], body)
    # the pairs that the receipt proposes, by atom
    pairs = [("present", "present"), ("bounds, guard of one parameter", "present"),
             ("bounds, guard of a split parameter", "split"), ("bounds, guard with cross terms", "cross"),
             ("bounds", "whole")]
    body = []
    for col, p in pairs:
        if p not in verdict:
            continue
        for atom in ["getOrElse", "ite", "cons", "append", "get", "mapFromEntries"]:
            for reading in ["normal", "raw"]:
                sel = [(k, r) for k, r in enumerate(data) if r["atom"] == atom and r["reading"] == reading]
                if not sel:
                    continue
                both = sum(r[col] == "true" and k not in verdict[p]["call"] for k, r in sel)
                lean_only = sum(r[col] == "true" and k in verdict[p]["call"] for k, r in sel)
                ts_only = sum(r[col] != "true" and k not in verdict[p]["call"] for k, r in sel)
                body.append([dict(lean_cols)[col], p, atom, reading, len(sel), both, lean_only, ts_only])
    table("The pairs of a reading and its prelude, by atom:",
          ["Lean", "prelude", "atom", "request", "applications", "both admit", "Lean admits, tsgo refuses",
           "tsgo accepts, Lean refuses"], body)
    # the answer's type where both admit
    body = []
    for p in verdict:
        both = [k for k, r in enumerate(data) if r["bounds"] == "true" and k not in verdict[p]["call"]]
        up = sum(k in verdict[p]["tsgo below Lean"] for k in both)
        down = sum(k in verdict[p]["Lean below tsgo"] for k in both)
        body.append([p, len(both), up, down])
    table("The call's type, where the match by bounds and tsgo both admit:",
          ["prelude", "applications", "tsgo's type is not assignable to Lean's answer",
           "Lean's answer is not assignable to tsgo's type"], body)
    # examples of each disagreement of the proposed pairs
    print("\nExamples (at most 6 of each kind):")
    for col, p in pairs:
        if p not in verdict:
            continue
        for kind, pick in [("Lean admits, tsgo refuses", lambda r, k: r[col] == "true" and k in verdict[p]["call"]),
                           ("tsgo accepts, Lean refuses", lambda r, k: r[col] != "true" and k not in verdict[p]["call"])]:
            hits = [(k, r) for k, r in enumerate(data) if pick(r, k)]
            if not hits:
                continue
            print(f"\n- {dict(lean_cols)[col]} against prelude {p}: {kind}: {len(hits)}")
            seen = collections.Counter()
            shown = 0
            for k, r in hits:
                if seen[(r["atom"], r["reading"])] >= 2 or shown >= 6:
                    continue
                seen[(r["atom"], r["reading"])] += 1
                shown += 1
                print(f"    - a{k}: {r['atom']}{r['arguments']} [{r['reading']}]; lower bounds: {r['lower bounds']}")
    for p in verdict:
        for role in ["tsgo below Lean", "Lean below tsgo"]:
            hits = [k for k, r in enumerate(data) if r["bounds"] == "true" and k not in verdict[p]["call"]
                    and k in verdict[p][role]]
            if hits:
                print(f"\n- prelude {p}, {role} fails: {len(hits)}")
                for k in hits[:4]:
                    r = data[k]
                    print(f"    - a{k}: {r['atom']}{r['arguments']} [{r['reading']}]; Lean answers {r['bounds answer']}")


if MODE == "gen":
    gen()
else:
    join()
