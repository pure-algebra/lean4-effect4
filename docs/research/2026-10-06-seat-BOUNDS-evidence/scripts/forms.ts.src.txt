import { Effect, Option, Ref } from "effect"
import { getOrElse, ite, append, cons, nil, some, none, pair } from "./prelude.ts"
declare const c: boolean
declare const o1: Option.Option<number>
declare const l1: ReadonlyArray<number>
declare const l2: ReadonlyArray<string>
declare const s: string
declare const n: number

export const O6_getOrElse_two = getOrElse(o1, "d")
export const T1_ite_two = ite(c, 1, "a")
export const L7_append_two = append(l1, l2)
// ---- the same three at variables, where no literal is inferred
export const O6v = getOrElse(o1, s)
