/- Seat BOUNDS: two omissions of each of two programs where only the match by bounds admits, with
the applications that decide: the arguments, both answers and the candidates. -/
namespace Bounds

def exampleOf (name : String) (table : RowTable) (p : P) : IO Unit := do
  if Api.wellTyped p table then
    let e := p.expandRefs
    let ss := Census.sites (nativeSignature table) [] e
    let effects : List (String × P) := ss.filterMap fun s =>
      (Census.replaceEff e s.path Census.neverProgram).map fun e' =>
        ("the node at " ++ showPath s.path ++ " (" ++ Census.ctorName (.eff s.node) ++ ")", e')
    let terms : List (String × P) := ss.flatMap fun s =>
      (Census.termSlots s.node).flatMap fun slot =>
        (Census.termParts slot.name slot.term).filterMap fun part =>
          (Census.replaceEff e s.path (slot.rebuild (part.rebuild Census.neverTerm))).map fun e' =>
            ("the term at " ++ showPath s.path ++ ", " ++ slot.name ++ " " ++ showPath part.path ++ " " ++ part.role, e')
    let mut shown := 0
    for (label, e') in effects ++ terms do
      if shown < 2 then
        if (Checker.check (sigN table) [] [] e').toOption.isNone &&
            ((runT (boundsRule true) (sigBN table) e').1).toOption.isSome then
          let (_, log) := runT presentRule (sigN table) e'
          let rows := (log.toList.flatMap (eventRows (sigN table))).filter fun (_, r) => r.verdictN == .admission
          shown := shown + 1
          IO.println s!"{name}, an omission of {label}:"
          for (_, r) in rows do
            IO.println s!"  {r.kind} {r.name}{r.inputs}: present {r.present}; bounds {r.bounds}; candidates {r.cands}"

#eval do
  for (name, p) in OCaml5.Truth.corpus do
    if name == "pQueueWake" || name == "pOptionSome" then
      exampleOf name (OCaml5.Truth.hostInputs name).1 p

end Bounds
