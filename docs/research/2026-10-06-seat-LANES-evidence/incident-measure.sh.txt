#!/usr/bin/env bash
# Read-only: what bun's global cache holds from the seat's session (it began 2026-10-06 17:19).
# It runs no bun, and it writes nothing under ~/.bun.
set -uo pipefail
cache="$HOME/.bun/install/cache"
echo "== every path under ~/.bun, to depth 2, that changed after the session began"
find "$HOME/.bun" -maxdepth 2 -newermt "2026-10-06 17:19:00" | sed "s|$HOME|~|"
echo "== the manifest cache of effect"
stat -f '%N  %z bytes  modified %Sm' -t '%Y-%m-%d %H:%M:%S' "$cache/b7c5cc8194df205f.npm" | sed "s|$HOME|~|"
strings "$cache/b7c5cc8194df205f.npm" | sed -n '2,3p'
echo "== the version folders of effect in the cache, newest first (none is from the session)"
ls -lT "$cache/effect" | awk 'NR>1 {print $6, $7, $8, $9, $10}' | sort -k4,4 -k1,1M -k2,2n -k3,3 | tail -3
echo "== node_modules above the worktree's root and above the scratch folder"
for d in "$1/node_modules" "$(dirname "$1")/node_modules" "$HOME/node_modules" "$2/node_modules" "$(dirname "$2")/node_modules"; do
  if [ -e "$d" ]; then echo "present: $d"; else echo "absent"; fi
done
