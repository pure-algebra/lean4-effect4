/-
Seat CENSUS, probe 2: the eliminator census. Every rule of `argTy`/`termTy`, `causeTy` and
`check` that inspects the type of a part is one entry: its name, the declaration that holds
it, what it reads, a type that it admits, two more that it admits one by one, and a function
that runs the rule at a given type of the variable `a0`. The table is the output of the last
`#eval`: each cell is what the tree's checker answers. The guards after it hold the claims of
the receipt, with one red control each.
-/
namespace Census.Elim

def sig0 : Sig := nativeSignature []
def call (name : String) (args : List Term) : Term := .app name (args.foldr .cons .nil)
def n (k : Nat) : Term := .lit (.nat k)
def str (t : String) : Term := .lit (.str t)
def v0 : Term := .var 0

/-- The outcome of a term at `a0 : T`: its type, or `REFUSED`. -/
def atTerm (t : Term) (T : Ty) : String :=
  match termTy sig0 [T] t with
  | some ty => showTy ty
  | none => "REFUSED"

/-- The outcome of a program at `a0 : T`: its three columns, or the refusal's rule. -/
def atEff (table : RowTable) (e : P) (T : Ty) : String :=
  match Checker.check (nativeSignature table) [T] [] e with
  | .ok ty => showEffTy ty
  | .error r => "REFUSED " ++ r.reason.head

def atCause (c : CauseTerm) (T : Ty) : String :=
  match causeTy sig0 [T] c with
  | some ty => showTy ty
  | none => "REFUSED"

structure Entry where
  name : String
  owner : String
  reads : String
  /-- Two types that the rule admits one by one. -/
  left : Ty
  right : Ty
  run : Ty → String

def unit_ : P := .succeed (.lit .unit)
def recA : Ty := .record [("_tag", false, .lit "A"), ("x", false, .nat)]
def recB : Ty := .record [("_tag", false, .lit "B"), ("x", false, .string)]
def tagA : Ty := .prod (.lit "A") .nat
def tagB : Ty := .prod (.lit "B") .string
def kNat : ServiceKey := ⟨⟨4⟩, ⟨4⟩⟩

/-- An external row with a closed request, for the row check at a closed row. -/
def closedTable : RowTable :=

     request := .string, answer := .nat, error := .never, cite := "seat CENSUS" }]

/-- `closeScope` reads a scope beside the exit: its row runs at `[T, scope]`. -/
def runEntry (e : Entry) (T : Ty) : String :=
  if e.name == "closeScope(scope, a0)" then
    match Checker.check sig0 [T, Ty.scope] [] (.withFiber (.closeScope (.var 1) v0)) with
    | .ok ty => showEffTy ty
    | .error r => "REFUSED " ++ r.reason.head
  else e.run T

def row (e : Entry) : String :=
  tsv [e.name, e.owner, e.reads, showTy e.left, runEntry e e.left, showTy e.right, runEntry e e.right,
    runEntry e (.union e.left e.right), runEntry e (Ty.join e.left e.right), runEntry e .never]

/-! ## The claims of the receipt, each with its red control -/

def T (t : Term) (env : TyEnv) : Option Ty := termTy sig0 env t
def E (e : P) (env : TyEnv) (table : RowTable := []) : Option EffTy :=
  (Checker.check (nativeSignature table) env [] e).toOption
def R (e : P) (env : TyEnv) : Option String :=
  (Checker.refusal (Checker.check sig0 env [] e)).map (·.reason.head)

-- 1. The four forms that go through a union's members and are total at `never`: the field read, the
--    field write, the tuple projection and `fst`/`snd`.
#guard T (.field .required v0 "x") [Ty.join recA recB] = some (Ty.join .nat .string)
#guard T (.field .required v0 "x") [.never] = some .never
#guard T (.recordSet v0 "x" (n 1)) [.never] = some .never
#guard T (.tupleAt v0 0) [.union (.prod .nat .string) (.tuple [.bool, .unit, .nat])] = some (Ty.join .nat .bool)
#guard T (.tupleAt v0 0) [.never] = some .never
#guard T (call "fst" [v0]) [.union (.prod .nat .string) (.prod .bool .unit)] = some (Ty.join .nat .bool)
#guard T (call "fst" [v0]) [.never] = some .never
-- red control: a member without the field refuses the whole read
#guard T (.field .required v0 "x") [Ty.join recA .nat] = none

-- 2. A fixed parameter takes each member of a union, and `never`; the answer is the declared one.
#guard T (call "concat" [v0, str "x"]) [.union (.lit "a") (.lit "b")] = some .string
#guard T (call "concat" [v0, str "x"]) [.never] = some .string
#guard T (call "concat" [v0, str "x"]) [.union (.lit "a") .nat] = none

-- 3. A template binds at the first member of a union and refuses the second; `never` binds nothing,
--    and an unbound parameter is `never` in the answer.
#guard T (call "get" [v0, n 0]) [.union (.list .nat) (.list .string)] = none
#guard T (call "get" [v0, n 0]) [.list (.union .nat .string)] = some (.option (.union .nat .string))
#guard T (call "get" [v0, n 0]) [.never] = some (.option .never)
#guard E (.perform .refGet v0) [.union (.refOf .nat) (.refOf .string)] = none
#guard E (.perform .refGet v0) [.never] = some ⟨.never, .never, Requirement.empty⟩

-- 4. `getOrElse` keeps its parameter at the first binding: it is not monotone in its first argument.
#guard T (call "getOrElse" [v0, n 9]) [.option .nat] = some .nat
#guard T (call "getOrElse" [v0, n 9]) [.option .never] = none
#guard T (call "getOrElse" [v0, n 9]) [.never] = some .nat
-- the joined templates are monotone at the same place
#guard T (call "append" [v0, call "cons" [n 1, call "nil" []]]) [.list .never] = some (.list .nat)

-- 5. A cell is invariant: the content's type binds at the cell, and a smaller cell refuses the write.
#guard E (.perform .refSet (call "pair" [v0, n 7])) [.refOf .nat] = some ⟨.refOf .nat, .never, Requirement.empty⟩
#guard R (.perform .refSet (call "pair" [v0, n 7])) [.refOf .never] = some "requestNotSubtype"
#guard E (.perform .refSet (call "pair" [v0, n 7])) [.never] = some ⟨.refOf .nat, .never, Requirement.empty⟩
#guard R (.perform (.refUpdateWith (call "succ" [.var 1])) v0) [.refOf .never] = some "resultNotSubtype"

-- 6. The forms that match a head or test an equality refuse `never` and a union.
#guard R (.awaitFiber v0 .joinEffect) [.never] = some "notFiber"
#guard R (.awaitFiber v0 .joinEffect) [.union (.fiberOf .nat .string) (.fiberOf .bool .nat)] = some "notFiber"
#guard R (.select v0 .bool unit_ unit_) [.never] = some "predicateNotBool"
#guard R (.select v0 .option unit_ unit_) [.never] = some "notSelectable"
#guard R (.select v0 .option unit_ unit_) [.union (.option .nat) (.option .string)] = some "notSelectable"
#guard R (.select v0 (.tag "A") unit_ unit_) [.never] = some "notSelectable"
#guard R (.restore v0 unit_) [.never] = some "maskRestoreExpected"
#guard R (.withFiber (.interruptAll v0 none)) [.list .never] = some "notFiber"
#guard R (.withFiber (.interruptAll (call "nil" []) none)) [] = some "notFiber"
#guard T (.fold none v0 (n 0) (.var 1)) [.never] = none
#guard T (.fold none v0 (n 0) (.var 1)) [.union (.list .nat) (.list .string)] = none
#guard T (call "causeIsFail" [v0]) [.never] = none
#guard T (call "causeIsFail" [v0]) [.union (.causeOf .nat) (.exitOf .unit .string)] = none
-- red control: the same forms at the type that they admit
#guard (E (.awaitFiber v0 .joinEffect) [.fiberOf .nat .string]).isSome
#guard (E (.select v0 .option unit_ unit_) [.option .nat]).isSome

-- 7. The two decisions that read members: the record's tag is total at `never`, the pair's tag is not.
#guard E (.select v0 (.recordTag "A") (.succeed (.var 1)) (.succeed (.var 1))) [.never] =
  some ⟨.never, .never, Requirement.empty⟩
#guard E (.select v0 (.recordTag "A") (.succeed (.var 1)) (.succeed (.var 1))) [Ty.join recA recB] =
  some ⟨Ty.join recA recB, .never, Requirement.empty⟩
#guard E (.select v0 (.tag "A") (.succeed (.var 1)) (.succeed (.var 1))) [Ty.join tagA tagB] =
  some ⟨Ty.join .nat tagB, .never, Requirement.empty⟩

-- 8. The comparisons by `Ty.sub` take a union member by member and take `never`.
#guard (E (.provideService kNat v0 unit_) [.never]).isSome
#guard (E (.withFiber (.awaitNewChildren v0)) [.never]).isSome
#guard (E (.withFiber (.awaitNewChildren v0)) [.union (.list (.fiberOf .nat .string)) (.list (.fiberOf .bool .nat))]).isSome
#guard E (.fail v0) [.never] = some ⟨.never, .never, Requirement.empty⟩
#guard E (.fail v0) [.union .nat .string] = some ⟨.never, .union .nat .string, Requirement.empty⟩

-- 9. A cursor or an accumulator without a stated type takes the initial value's type: a smaller
--    initial value then refuses the step.
#guard (E (.iterate none v0 (.lit (.bool true)) (call "succ" [.var 1]) (.lit .unit) unit_) [.nat]).isSome
#guard R (.iterate none v0 (.lit (.bool true)) (call "succ" [.var 1]) (.lit .unit) unit_) [.never] = some "stepNotCursor"
#guard (E (.iterate (some .nat) v0 (.lit (.bool true)) (call "succ" [.var 1]) (.lit .unit) unit_) [.never]).isSome
#guard T (.fold none (call "cons" [n 1, call "nil" []]) v0 (call "add" [.var 1, .var 2])) [.nat] = some .nat
#guard T (.fold none (call "cons" [n 1, call "nil" []]) v0 (call "add" [.var 1, .var 2])) [.never] = none
#guard T (.fold (some .nat) (call "cons" [n 1, call "nil" []]) v0 (call "add" [.var 1, .var 2])) [.never] = some .nat

#eval IO.println "probe 2: every guard above holds"
