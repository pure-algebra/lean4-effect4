#!/usr/bin/env python3
"""Seat CENSUS: each number of the receipt's prose, computed again from the rows and the logs.

    verify.py            check the receipt; exit 1 when a sentence with its numbers is not there
    verify.py --count    print the number of checks alone (the builder writes it into the receipt)

A check is a piece of the receipt's text whose numbers this script computes: the receipt must
hold that text (white space apart). A number that changes in the rows changes the text that is
looked for, so the receipt and the rows cannot differ in silence. The tables of the receipt
are written by `mdtables.py` from the same rows, and they are not checked here again. The
receipt is read from the first argument that is a path, or from the worktree's place for it.
"""
import collections, csv, os, re, sys

S = os.path.dirname(os.path.abspath(__file__))
W = '/Users/pooks/Dev/lean4-effect4-lower'
MAIN = '/Users/pooks/Dev/lean4-effect4'      # the coordinator's checkout: read only
RECEIPT = os.path.join(W, 'docs/research/2026-10-06-seat-CENSUS-receipt.md')
EVIDENCE = os.path.join(W, 'docs/research/2026-10-06-seat-CENSUS-evidence')
PLAN = os.path.join(W, 'docs/research/2026-10-06-type-slicing-plan.md')
BRIEF = os.path.join(W, 'docs/research/2026-10-05-claude-lead/briefs/seat-census-brief.md')
COUNT = '--count' in sys.argv
for a in sys.argv[1:]:
    if a != '--count': RECEIPT = a

def read(name):
    rows = list(csv.reader(open(os.path.join(S, 'out', name), newline=''), delimiter='\t', quoting=csv.QUOTE_NONE))
    bad = [r for r in rows[1:] if len(r) != len(rows[0])]
    assert not bad, (name, 'malformed rows', len(bad))
    return [dict(zip(rows[0], r)) for r in rows[1:]]
def log(name): return open(os.path.join(S, name + '.log')).read()
def norm(s): return re.sub(r'\s+', ' ', s).strip()
WORD = {1: 'one', 2: 'two', 3: 'three', 4: 'four', 5: 'five', 6: 'six', 7: 'seven', 8: 'eight', 9: 'nine', 10: 'ten', 12: 'twelve'}
def one(pattern, text, cast=int):
    m = re.search(pattern, text)
    assert m, pattern
    return cast(m.group(1)) if m.lastindex == 1 else tuple(cast(g) for g in m.groups())

needles = []      # (the text looked for in the receipt, what it rests on)
facts = []        # (a statement checked on the rows alone, its truth)
def need(text, source): needles.append((norm(text), source))
def fact(text, ok): facts.append((text, bool(ok)))

MODULES = ('pQueueW', 'pQueueF', 'pQueueI', 'pQueueM', 'pQueueOr', 'pSemaphore', 'pPool')
def is_module(p): return any(p.startswith(m) for m in MODULES)
BAD = ('larger', 'bothWays', 'unrelated')
def bad(r): return any(r[c] in BAD for c in ('answer', 'error', 'requires'))
def refused(r): return r['admitted'] != 'true'
def node_error(r): return r['nodeType'].strip('<>').split(' ; ')[1]

gf, tf = read('gen-folds.tsv'), read('truth-folds.tsv')
gs, ts = read('gen-sites.tsv'), read('truth-sites.tsv')
gt, tt = read('gen-terms.tsv'), read('truth-terms.tsv')
gp, tp = read('gen-programs.tsv'), read('truth-programs.tsv')
folds, sites, terms = gf + tf, gs + ts, gt + tt
others_f = [r for r in tf if not is_module(r['program'])]
modules_f = [r for r in tf if is_module(r['program'])]

# ---------------------------------------------------------------- the corpora (sections 3 and 4)
g_programs, g_admitted = one(r'gen: programs (\d+); admitted (\d+);', log('probe1gen'))
t_programs, t_admitted = one(r'truth: programs (\d+); admitted (\d+);', log('probe1truth'))
g_sites, g_rows = one(r'gen: sites (\d+); fold rows (\d+);', log('probe1gen'))
t_sites, t_rows = one(r'truth: sites (\d+); fold rows (\d+);', log('probe1truth'))
g_terms = one(r'gen: term addresses (\d+);', log('probe1termgen'))
t_terms = one(r'truth: term addresses (\d+);', log('probe1termtruth'))
fact('the rows hold the sites and the folds that the logs count', (len(gs), len(ts), len(gf), len(tf), len(gt), len(tt)) == (g_sites, t_sites, g_rows, t_rows, g_terms, t_terms))
fact('the programs files hold the admitted programs', (len(gp), len(tp)) == (g_admitted, t_admitted))
for label in ('gen', 'truth'):
    text = log('probe1' + label)
    fact(label + ': the site walk agrees with foldMapAt_eff, and the second walk with the first',
         'the site walk agrees with foldMapAt_eff on every program: true' in text and 'envAt agrees with the site walk on every program: true' in text)
n_eff, n_term = len(sites), len(terms)
mod_programs = sorted(set(r['program'] for r in tp if is_module(r['program'])))
n_mod, n_others = len(mod_programs), len(tp) - len(mod_programs)
mod_eff = sum(int(r['effectNodes']) for r in tp if is_module(r['program']))
others_eff = sum(int(r['effectNodes']) for r in tp if not is_module(r['program']))
largest = max(tp, key=lambda r: int(r['effectNodes']))
need(f'| 400 | {g_admitted} | {g_sites} | {g_terms} |', 'probe1gen.log, probe1termgen.log')
fact('the generated corpus has 400 programs', g_programs == 400)
need(f'| {t_programs} | {t_admitted} | {t_sites} | {t_terms} |', 'probe1truth.log, probe1termtruth.log')
need(f'| `Probe1Gen`, `Probe1Truth` | the effect addresses: {g_sites} and {t_sites} sites; {g_rows} and {t_rows} fold rows |', 'the two logs')
need(f'| `Probe1TermGen`, `Probe1TermTruth` | the term addresses: {g_terms} and {t_terms} |', 'the two logs')
need(f'Its {g_admitted} admitted programs hold {sum(int(r["effectNodes"]) for r in gp)} effect nodes, under two each. '
     f'The largest holds {max(int(r["effectNodes"]) for r in gp)}, and no environment in it has more than {max(int(r["level"]) for r in gs)} entry.', 'gen-programs.tsv, gen-sites.tsv')
fact('the generated programs hold under two effect nodes each', sum(int(r['effectNodes']) for r in gp) < 2 * len(gp))
need(f"The truth lane's {n_mod} programs of the modules (the Queue's, Semaphore's and Pool's) hold {mod_eff} of its {t_sites} effect addresses. "
     f"The largest, `{largest['program']}`, holds {largest['effectNodes']}, and its largest environment has {max(int(r['level']) for r in ts if r['program'] == largest['program'])} entries.",
     'truth-programs.tsv, truth-sites.tsv')
fact('the largest environment of the truth lane is in its largest program', max(int(r['level']) for r in ts) == max(int(r['level']) for r in ts if r['program'] == largest['program']))
refs_g = one(r'gen: programs \d+; admitted \d+; admitted with a reference in the stored program: (\d+)', log('probe1gen'))
refs_t = one(r'truth: programs \d+; admitted \d+; admitted with a reference in the stored program: (\d+)', log('probe1truth'))
truth_src = open(os.path.join(W, 'harness/truth/Truth.lean')).read().split('\n')

for line in truth_src:
    m = re.match(r'def (\w+)', line)
    if m: cur = m.group(1)
    if '.ref [' in line: ref_defs.add(cur)
fact('no admitted generated program holds a layer reference, and one of the truth lane does: pDiamond', (refs_g, refs_t, ref_defs) == (0, 1, {'pDiamond'}))
need('One program of the truth lane does (`pDiamond`).', 'probe1truth.log, harness/truth/Truth.lean')
fact('lines 97 to 883 of Truth.lean run from the namespace to the end of the corpus',
     truth_src[96] == 'namespace OCaml5.Truth' and truth_src[882].endswith('("pPoolClosing", pPoolClosing)]') and "sed -n '97,883p'" in open(os.path.join(S, 'build.sh')).read())
need('The probes take its lines 97 to 883 as text', 'build.sh, harness/truth/Truth.lean')
thrifty = set(r['program'] for r in tp if int(r['effectNodes']) > 40)
fact('the programs of more than 40 effect nodes are the programs of the modules', thrifty == set(mod_programs) and not any(int(r['effectNodes']) > 40 for r in gp))
need(f'A program of more than 40 effect nodes runs the row form\'s full fold at three kinds of site only. Those programs are the {n_mod} of the modules.', 'truth-programs.tsv, LibRun.lean')
fact('LibRun.lean runs a program of more than 40 effect nodes thrifty', 'small : Nat := 40' in open(os.path.join(S, 'LibRun.lean')).read())
selftest = open(os.path.join(S, 'SelfTest.tail.lean')).read()
n_known = len(re.findall(r'(?m)^def t\d+ : P', selftest))
need(f'holds {WORD[n_known]} programs with answers known by hand', 'SelfTest.tail.lean')
need(f'| `SelfTest` | the probe\'s own test: {WORD[n_known]} programs with answers known by hand', 'SelfTest.tail.lean')
fact('the base build completed', 'Build completed successfully (1031 jobs).' in open(os.path.join(S, 'base-build.log')).read())
need('`Build completed successfully (1031 jobs)`', 'base-build.log')

     all(not refused(r) and r['answer'] == 'same' and not bad(r) for r in in_plan))
fact('inside the wider premise every fold A is admitted with the answer identical and no column larger',
     all(not refused(r) and r['answer'] == 'same' and not bad(r) for r in in_wide))
need(f'- **Inside the plan\'s domain: {len(in_plan)} folds, {sum(map(refused, in_plan))} refused, the answer identical at all of them.**', 'the folds files')
need(f'- **The wider premise: {len(in_wide)} folds, {sum(map(refused, in_wide))} refused, the answer identical at all of them.**', 'the folds files')
fork_first = sum(1 for r in tf if r['fold'] == 'N' and r['firstClosed'] == 'ActionTerm.fork.0')
need(f'It adds {len(in_wide) - len(in_plan)} folds under a closed edge. At {fork_first} sites of the truth lane the first closed edge above is a fork.', 'the folds files')
rest_ref, rest_same = sum(map(refused, rest)), same_answer(rest)
rest_changed = len(rest) - rest_ref - rest_same
need(f'- **Outside the wider premise: {len(rest)} folds, {rest_ref} refused.**', 'the folds files')
need(f'{rest_same} leave the root\'s answer the same, and {rest_changed} change it.', 'the folds files')
fact('outside the wider premise a changed answer is a smaller one', all(r['answer'] in ('same', 'smaller') for r in rest if not refused(r)))
mod_a = fold_a(modules_f)

mod_wide = sum(1 for r in mod_a if r['columnOpen'] == 'true' or node_error(r) == 'never')
need(f'The plan\'s domain holds {mod_plan} of their {mod_eff} addresses. The wider premise holds {mod_wide}.', 'truth-folds.tsv')
need(f'of the {len(rest)} folds outside it, {rest_ref + rest_changed} are refused or change the answer.', 'the folds files')
need(f'None meets one of the {len(rest)} folds.', 'the folds files')
fact('the four refusals of fold A are getOrElse in pOptionSome and pOptionNone',
     sorted((r['program'], r['culprit']) for r in A if refused(r)) == sorted([(p, 'getOrElse(option<never>, nat)') for p in ('pOptionNone', 'pOptionSome')] * 2))

# the flow of a folded error
flow = []
for label in ('gen', 'truth'):
    for r in read(label + '-flow.tsv'): r['corpus'] = label; flow.append(r)
by_fold = collections.OrderedDict()
for r in flow: by_fold.setdefault((r['corpus'], r['program'], r['address']), []).append(r)
def summary(rs): return [r for r in rs if r['kind'] == 'summary'][0]
flow_in = {k: v for k, v in by_fold.items() if summary(v)['columnOpen'] == 'true'}
flow_out = {k: v for k, v in by_fold.items() if summary(v)['columnOpen'] != 'true'}
fact('the flow probe runs the folds of a node that can fail', len(by_fold) == sum(1 for r in A if node_error(r) != 'never'))
fact('in the plan\'s domain no environment entry and no site\'s answer changes',
     all(summary(v)['position'] == 'no environment entry changes' and not any(r['kind'] in ('use', 'answer') for r in v) for v in flow_in.values()))
env_changes = sum(1 for v in flow_out.values() if summary(v)['position'] == 'an environment entry changes')
uses = collections.defaultdict(lambda: [set(), set()])
for k, v in flow_out.items():
    for r in v:
        if r['kind'] == 'use': uses[r['position']][0 if summary(v)['admitted'] == 'true' else 1].add(k)
need(f'each of the {len(by_fold)} folds of a node that can fail', 'the flow files')
need(f'- In the plan\'s domain: {len(flow_in)} folds. No environment entry and no site\'s answer changes.', 'the flow files')
need(f'- Outside it: {len(flow_out)} folds. {env_changes} change an environment entry, and {WORD[len(uses)]} positions read a changed variable:', 'the flow files')
LABEL = {'causeError.0': "`causeError`'s argument", 'tagIs.1': "`tagIs`'s second argument", 'causeIsFail.0': "`causeIsFail`'s argument",
         'succeed.value': 'a `succeed` value', 'eq.0': "`eq`'s first argument", 'isSome.0': "`isSome`'s argument", 'getOrElse.0': "`getOrElse`'s first argument"}
fact('the positions that read a changed variable are the seven of the table', set(uses) == set(LABEL))
for pos, (adm, ref) in uses.items():
    need(f'| {LABEL.get(pos, pos)} | {len(adm)} | {len(ref)} |', 'the flow files')
refusing = [p for p, (adm, ref) in uses.items() if ref and not adm]
fact('the positions that only refused folds reach are getOrElse.0 and isSome.0', sorted(refusing) == ['getOrElse.0', 'isSome.0'])
need(f'One position refuses, in all {len(uses["getOrElse.0"][1])} folds that reach it: `getOrElse`\'s first argument.', 'the flow files')
need(f'| `Probe1Flow` | where a folded error flows: {len(by_fold)} folds of a node that can fail |', 'the flow files')
need(f'on the {len(flow_in)} folds of a failing node in the plan\'s domain', 'the flow files')
premise = log('probe4premise')
pm = [tuple(int(x) for x in m) for m in re.findall(
    r'programs (\d+); closed masks (\d+); masks in the wider premise (\d+): admitted (\d+), with the full program\'s answer (\d+), '
    r'with error and requirement no larger (\d+); pairs a ≤ b of them (\d+), with error and requirement ordered (\d+); masks outside the premise (\d+), refused (\d+)', premise)]
assert len(pm) == 2, premise
pm = [sum(x) for x in zip(*pm)]
p_programs, p_masks, p_wide, p_adm, p_answer, p_cols, p_pairs, p_ordered, p_out, p_out_ref = pm
fact('every mask of the wider premise is admitted with the full answer, and every pair of them is ordered', p_wide == p_adm == p_answer == p_cols and p_pairs == p_ordered)
need(f'It is tested by {len(in_wide)} single folds and by {p_wide} whole masks (section 9.3), with no refusal and no changed answer.', 'the folds files, probe4premise.log')
need(f'{len(in_wide)} single folds and {p_wide} whole masks meet it.', 'the folds files, probe4premise.log')
need(f'The plan\'s own domain covers {len(in_plan)} of those folds.', 'the folds files')

# ---------------------------------------------------------------- the row form (5.4)
layer = [r for r in folds if r['fold'] == 'N' and r['inLayer'] == 'true']

fact('the row\'s own type is the assumption\'s at every site, and every answer is its own normal form', all(r['rowLocal'] == 'same' and r['answerNormal'] == 'true' for r in sites))
need(f'And the row\'s own node has the type `⟨A, never, ∅⟩` at all {n_eff} sites.', 'the sites files')

need(f'Its own type is the assumption\'s at all {n_eff} sites.', 'the sites files')
need(f'It checks at all {n_eff} sites with the assumption\'s own type.', 'the sites files')
need(f'The two agree at all {len(both)} addresses where both ran.', 'the folds files')

# ---------------------------------------------------------------- fold N (5.5 and 6)

# the rules that refuse: `family` of mdtables.py, read from its text so that the two scripts hold one definition

need(f'Each refusal names one of {len(fam)} rules. Uniform eliminators would remove {k1} of the {total_ref} refusals. {k2} come from a rule that takes its expectation from a part, and {k3} from a cell\'s invariance.', 'the folds and terms files')
need(f'The {len(fam)} rules are of three kinds', 'the folds and terms files')
need(f'| `Probe1Small` | one smallest program for each of the {len(fam)} rules that refuse |', 'probe1small.log')
