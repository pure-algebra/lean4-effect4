import Effect4.Laws
import Test.All
open Lean Meta

/-- The declaration that a person wrote and that owns a constant: its longest prefix with a
declaration range. -/
partial def ownerOf (env : Environment) (ranges : NameSet) (n : Name) : Name :=
  if ranges.contains n then n
  else match n with
    | .str p _ => if p.isAnonymous then n else ownerOf env ranges p
    | .num p _ => if p.isAnonymous then n else ownerOf env ranges p
    | .anonymous => n

#eval show MetaM Unit from do
  let env ← getEnv
  let targets : List Name := [`Effect4.Laws.Program.Template, `Effect4.Data.Constructive]
  let modOf (n : Name) : Option Name := (env.getModuleIdxFor? n).map fun i => env.header.moduleNames[i.toNat]!
  -- the declarations that a person wrote
  let mut ranges : NameSet := {}
  for (n, _) in env.constants.toList do
    if (← findDeclarationRanges? n).isSome then ranges := ranges.insert n
  -- callers by owner
  let mut callers : Std.HashMap Name NameSet := {}
  for (n, ci) in env.constants.toList do
    let o := ownerOf env ranges n
    let used := ci.getUsedConstantsAsSet
    for u in used.toList do
      let ou := ownerOf env ranges u
      if ou != o then
        callers := callers.insert ou ((callers.getD ou {}).insert o)
  let cands := ranges.toList.filter fun n => (modOf n).any (targets.contains ·)
  let candSet : NameSet := cands.foldl (·.insert ·) {}
  -- live: a caller outside the candidates, then closed under "called by a live candidate"
  let mut live : NameSet := {}
  for c in cands do
    if ((callers.getD c {}).toList.any fun k => !candSet.contains k) then live := live.insert c
  let mut changed := true
  while changed do
    changed := false
    for c in cands do
      if !live.contains c ∧ ((callers.getD c {}).toList.any fun k => live.contains k) then
        live := live.insert c; changed := true
  let dead := (cands.filter fun c => !live.contains c).toArray.qsort (fun a b => a.toString < b.toString)
  IO.println s!"candidates {cands.length}; with no caller chain from another module {dead.size}"
  for d in dead do
    IO.println s!"{(modOf d).getD .anonymous}\t{d}\tcallers:{((callers.getD d {}).toList.map toString)}"
