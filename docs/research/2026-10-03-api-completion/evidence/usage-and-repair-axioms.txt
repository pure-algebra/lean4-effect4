Effect4.Run.nextControl_spec (s : Run) (decision : Api.Decision) :
  s.nextControl = some decision ↔
    s.machine.stuck (Prim Program.EffName Program.EffThunk Machine.Val Machine.Err Machine.Defect FiberId Machine.Ann)
          (FrameFiber Program.EffName Program.EffThunk Machine.Val Machine.Err Machine.Defect FiberId Machine.Ann)
          (FrameEvent Program.EffName Program.EffThunk Machine.Val Machine.Err Machine.Defect FiberId Machine.Ann) =
        none ∧
      (decision = Api.flush ∧ s.work.queued ≠ [] ∨
        ∃ fiber,
          decision = Machine.RunDecision.evaluate fiber ∧ s.work.queued = [] ∧ s.work.runnable.head? = some fiber)
