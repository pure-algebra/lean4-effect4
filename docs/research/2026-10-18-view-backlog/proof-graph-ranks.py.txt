# Replicates the scratch timer's graph (graphOf k) and Graph.ranks / Graph.items, to count items per rank.
import json, sys
r = json.load(open(sys.argv[1])); plan = r['plan']
def graph(k):
    reqs = plan['requirements'][:k]
    out = {}
    for q in reqs: out['req:'+q['id']] = [t['name'] for t in q.get('top', [])] + [t['name'] for t in q.get('placed', [])]
    for n in plan['nodes']: out[n['name']] = n['broughtIn']['nearest']
    reqIds = ['req:'+q['id'] for q in reqs]
    seen = set(); todo = list(reqIds)
    while todo:
        x = todo.pop(0)
        if x in seen: continue
        seen.add(x); todo = out.get(x, []) + todo
    ths = [n['name'] for n in plan['nodes'] if n['name'] in seen]
    indeg = {}
    for u in ths:
        for v in out[u]:
            if v in seen: indeg[v] = indeg.get(v, 0) + 1
    order = []; ready = [u for u in ths if indeg.get(u, 0) == 0]
    while ready:
        v = ready.pop(0); order.append(v)
        for w in out[v]:
            if w in seen:
                indeg[w] -= 1
                if indeg[w] == 0: ready.append(w)
    below = {}
    for u in reversed(order):
        s = set()
        for v in out[u]:
            if v in seen: s.add(v); s |= below.get(v, set())
        below[u] = s
    allk = reqIds + order; index = {k_: i for i, k_ in enumerate(allk)}
    edges = []
    for rq in reqIds:
        for t in out[rq]:
            if t in index: edges.append((index[rq], index[t]))
    for u in order:
        ks = [v for v in out[u] if v in seen]
        for v in ks:
            if not any(w != v and v in below.get(w, set()) for w in ks): edges.append((index[u], index[v]))
    return allk, edges
for k in [1, 3, 4, 6, 14]:
    nodes, edges = graph(k)
    rk = [0]*len(nodes)
    preds = {}
    for (a, b) in edges:
        if a < b: preds.setdefault(b, []).append(a)
    for v in range(len(nodes)): rk[v] = max([rk[u]+1 for u in preds.get(v, [])], default=0)
    per = {}
    for v in range(len(nodes)): per[rk[v]] = per.get(rk[v], 0) + 1
    for (a, b) in edges:
        if a < b:
            for r_ in range(rk[a]+1, rk[b]): per[r_] = per.get(r_, 0) + 1
    print(f'k={k}: {len(nodes)} nodes, {len(edges)} edges, {sum(per.values())} items in {len(per)} ranks; widest rank {max(per.values())} items (rank {max(per, key=per.get)})')
