def strs (j : Json) : List String :=
  match j.getArr? with
  | .ok a => a.toList.filterMap fun x => (x.getStr?).toOption
  | .error _ => []

def names (j : Json) : List String :=
  match j.getArr? with
  | .ok a => a.toList.filterMap fun x => ((x.getObjValD "name").getStr?).toOption
  | .error _ => []

/-- The proof graph the d3-dag page draws by default, as a `Graph`. -/
def graphOf (k : Nat) (j : Json) : Graph := Id.run do
  let plan := j.getObjValD "plan"
  let reqs := ((plan.getObjValD "requirements").getArr?.toOption.getD #[]).extract 0 k
  let nodes := (plan.getObjValD "nodes").getArr?.toOption.getD #[]
  let mut out : Std.HashMap String (List String) := {}
  for r in reqs do
    let id := ((r.getObjValD "id").getStr?).toOption.getD ""
    out := out.insert ("req:" ++ id) (names (r.getObjValD "top") ++ names (r.getObjValD "placed"))
  for n in nodes do
    let name := ((n.getObjValD "name").getStr?).toOption.getD ""
    out := out.insert name (strs ((n.getObjValD "broughtIn").getObjValD "nearest"))
  let reqIds := reqs.toList.map fun r => "req:" ++ ((r.getObjValD "id").getStr?).toOption.getD ""
  let mut seen : Std.HashSet String := {}
  let mut todo := reqIds
  for _ in [0:1000000] do
    match todo with
    | [] => break
    | x :: rest =>
      todo := rest
      unless seen.contains x do
        seen := seen.insert x
        todo := (out.getD x []) ++ todo
  let ths := (nodes.toList.filterMap fun n => ((n.getObjValD "name").getStr?).toOption).filter seen.contains
  let mut indeg : Std.HashMap String Nat := {}
  for u in ths do
    for v in out.getD u [] do
      if seen.contains v then indeg := indeg.insert v (indeg.getD v 0 + 1)
  let mut order : Array String := #[]
  let mut ready := ths.filter (indeg.getD · 0 == 0)
  for _ in [0:1000000] do
    match ready with
    | [] => break
    | v :: rest =>
      ready := rest
      order := order.push v
      for w in out.getD v [] do
        if seen.contains w then
          let d := indeg.getD w 0 - 1
          indeg := indeg.insert w d
          if d == 0 then ready := ready ++ [w]
  let mut below : Std.HashMap String (Std.HashSet String) := {}
  for u in order.reverse do
    let mut s : Std.HashSet String := {}
    for v in out.getD u [] do
      if seen.contains v then
        s := s.insert v
        for w in (below.getD v {}).toList do s := s.insert w
    below := below.insert u s
  let all := reqIds.toArray ++ order
  let index : Std.HashMap String Nat := all.zipIdx.foldl (fun m (k, i) => m.insert k i) {}
  let mut edges : Array GEdge := #[]
  for r in reqIds do
    for t in out.getD r [] do
      if let some j := index.get? t then
        edges := edges.push { key := s!"{r}→{t}", src := index.getD r 0, dst := j }
  for u in order do
    let ks := (out.getD u []).filter seen.contains
    for v in ks do
      let implied := ks.any fun w => w != v && (below.getD w {}).contains v
      unless implied do
        edges := edges.push { key := s!"{u}→{v}", src := index.getD u 0, dst := index.getD v 0 }
  return { nodes := all.map fun k => { key := k, line1 := (k.splitOn ".").getLast!, line2 := "" }, edges }

def measure (k : Nat) (path : String) : IO Unit := do
  let say (s : String) : IO Unit := do IO.println s; (← IO.getStdout).flush
  let t0 ← IO.monoMsNow
  let text ← IO.FS.readFile path
  let .ok j := Json.parse text | throw (IO.userError "parse")
  let g ← IO.lazyPure fun _ => graphOf k j
  let n := g.nodes.size + g.edges.size
  let t1 ← IO.monoMsNow
  say s!"k={k} graph: {g.nodes.size} nodes, {g.edges.size} edges ({n}); read and built in {t1 - t0} ms"
  let rk ← IO.lazyPure fun _ => Graph.ranks g
  let t2 ← IO.monoMsNow
  say s!"ranks: {rk.foldl max 0 + 1} ranks in {t2 - t1} ms"
  let os ← IO.lazyPure fun _ => Graph.orders g rk
  let t3 ← IO.monoMsNow
  say s!"orders: {os.foldl (· + ·.length) 0} items in {os.length} ranks in {t3 - t2} ms"
  let (placed, routes, w, h, sum) ← IO.lazyPure fun _ =>
    let l := g.layout
    (l.placed.size, l.routes.size, l.width, l.height, l.placed.foldl (fun s p => s + p.x.natAbs + p.y.natAbs) 0)
  let t4 ← IO.monoMsNow
  say s!"layout: {placed} placed, {routes} routes, {w}x{h} px, checksum {sum}, in {t4 - t3} ms"

end Tools.ScratchMeasure

/-- The layout's steps one by one, as `Graph.layout` calls them. -/
def stages (k : Nat) (path : String) : IO Unit := do
  let say (s : String) : IO Unit := do IO.println s; (← IO.getStdout).flush
  let .ok j := Json.parse (← IO.FS.readFile path) | throw (IO.userError "parse")
  let g ← IO.lazyPure fun _ => graphOf k j
  let rk ← IO.lazyPure fun _ => Graph.ranks g
  let its ← IO.lazyPure fun _ => Graph.items g rk
  let width (k : Key) : Nat := ((its.find? (·.key == k)).map (·.width)).getD GAP
  let os ← IO.lazyPure fun _ => Graph.orders g rk
  let ls ← IO.lazyPure fun _ => Graph.links g rk
  let ups (k : Key) := ls.filterMap fun (a, b) => if b == k then some a else none
  let downs (k : Key) := ls.filterMap fun (a, b) => if a == k then some b else none
  let isPoint (k : Key) : Bool := ((its.find? (·.key == k)).map (·.node.isNone)).getD false
  say s!"k={k}: {g.nodes.size} nodes, {g.edges.size} edges, {its.length} items, {ls.length} links"
  let t0 ← IO.monoMsNow
  let cs ← IO.lazyPure fun _ => Place.conflicts os ls isPoint
  let t1 ← IO.monoMsNow
  say s!"conflicts: {cs.length} marked in {t1 - t0} ms"
  let c ← IO.lazyPure fun _ => Place.centres os width ups downs ls isPoint
  let t2 ← IO.monoMsNow
  say s!"centres (with conflicts again): {c.size} items in {t2 - t1} ms"
  let n ← IO.lazyPure fun _ => (os.flatten.map fun k => (width k, isPoint k)).length
  let t3 ← IO.monoMsNow
  say s!"one width and isPoint lookup per item: {n} in {t3 - t2} ms"

end Tools.ScratchMeasure
