#!/usr/bin/env python3
"""Seat GAP: of the omissions that candidate N with uniform eliminators refuses, how many hold
a cell constructor (mk, get, set) in the sketch?  Run: python3 d1_cells.py [size, default 5]"""
import itertools, sys
from gap_checker import (terms, check, STATIC, N_UNIFORM, N_AS_IS, subterms, replace, tshow)

def main(n):
    closed = [e for k in range(1, n + 1) for e in terms(k, 0, False)]
    admitted = [e for e in closed if check(STATIC, [], e) is not None]
    for label, mode in [("N, uniform eliminators", N_UNIFORM), ("N, eliminators as they are", N_AS_IS)]:
        refused = with_cell = 0
        without = []
        for e in admitted:
            pos = [p for p, _ in subterms(e) if p != ()]
            folds = [[p] for p in pos] + [[p, q] for p, q in itertools.combinations(pos, 2)
                                         if p[:len(q)] != q and q[:len(p)] != p]
            for fold in folds:
                sk = e
                for p in fold:
                    sk = replace(sk, p, ("?", "".join(map(str, p))))
                if check(mode, [], sk) is None:
                    refused += 1
                    if any(s[0] in ("mk", "get", "set") for _, s in subterms(sk)):
                        with_cell += 1
                    else:
                        without.append(sk)
        print(f"{label}: refused {refused}; of them with a cell constructor {with_cell}; "
              f"without {len(without)}" + ("" if not without else "   e.g. " + tshow(without[0])))

main(int(sys.argv[1]) if len(sys.argv) > 1 else 5)
